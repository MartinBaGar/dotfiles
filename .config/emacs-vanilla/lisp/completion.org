* Minibuffer settings
Vanilla Emacs settings tuned for Vertico. Bound via =use-package emacs=
so they get the same =:custom= syntax as everything else, even though
there's nothing to fetch.
#+begin_src emacs-lisp
(use-package emacs
  :ensure nil ; built-in
  :custom
  (context-menu-mode t)
  (enable-recursive-minibuffers t)
  (read-extended-command-predicate #'command-completion-default-include-p)
  (minibuffer-prompt-properties
   '(read-only t cursor-intangible t face minibuffer-prompt)))
#+end_src

(use-package cape
  :bind (("C-c p" . cape-prefix-map)
         ("C-c p e" . my/cape-elisp-everywhere))
  :init
  (setq text-mode-ispell-word-completion nil)
  
  (defalias 'my/cape-dabbrev-prose
    (cape-capf-prefix-length (cape-capf-accept-all #'cape-dabbrev) 5))

  (defalias 'my/cape-dict-prose
    (cape-capf-prefix-length #'cape-dict 5))

  (defalias 'my/cape-dabbrev-code
    (cape-capf-inside-faces
     (cape-capf-prefix-length #'cape-dabbrev 5)
     'font-lock-comment-face 'font-lock-doc-face 'font-lock-string-face))

  (defalias 'my/cape-file-in-string
    (cape-capf-inside-string #'cape-file))

  ;; Optional: interactive Elisp completion anywhere (e.g., Org prose)
  (defalias 'my/cape-elisp-everywhere
    (cape-capf-interactive #'elisp-completion-at-point))

  ;; 2. Global defaults (used as fallback when buffer-local list ends with `t`)
  (add-hook 'completion-at-point-functions #'tempel-complete 10)
  (add-hook 'completion-at-point-functions #'cape-elisp-block 20)

  ;; 3. Programming modes: file paths only in strings, dabbrev only in comments/strings
  (add-hook 'prog-mode-hook
            (lambda ()
              (add-hook 'completion-at-point-functions #'my/cape-file-in-string 30 t)
              (add-hook 'completion-at-point-functions #'my/cape-dabbrev-code 90 t)))

  ;; 4. Text / Org modes: merge dabbrev + dict in prose, skip inside #+begin_src blocks
  (defun my/org-or-text-prose-capf ()
    (unless (and (derived-mode-p 'org-mode) (org-in-src-block-p))
      (funcall (cape-capf-sort
                (cape-capf-super #'my/cape-dabbrev-prose #'my/cape-dict-prose)))))

  (add-hook 'text-mode-hook
            (lambda ()
              (add-hook 'completion-at-point-functions #'cape-file 30 t)
              (add-hook 'completion-at-point-functions #'my/org-or-text-prose-capf 90 t))))
#+end_src
