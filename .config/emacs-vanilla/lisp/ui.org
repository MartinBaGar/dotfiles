* Fonts
Base faces only. The org-specific fixed-pitch overrides (tables, code
blocks, etc) live in org-config.org, next to the rest of the org
appearance settings.
#+begin_src emacs-lisp
  ;; Fixed-pitch (code, and the "default" face everything falls back to)
  (set-face-attribute 'default nil
                      :family "Iosevka"
                      :height 120        ; height is in 1/10 pt, so 120 = 12pt
                      :weight 'medium)

  ;; fixed-pitch should usually just mirror default
  (set-face-attribute 'fixed-pitch nil
                      :family "Iosevka"
                      :height 120
                      :weight 'medium)

  ;; Variable-pitch (org-mode prose, markdown, UI elements)
  (set-face-attribute 'variable-pitch nil
                      :family "Montserrat"
                      :height 130
                      :weight 'semi-bold)

  ;; 1. The main bar: inherit the background color from the inactive mode-line
  (set-face-attribute 'tab-bar nil 
                      :background 'unspecified 
                      :inherit 'mode-line-inactive 
                      :box nil)

  ;; 2. Active tab: inherit the exact foreground color of default text
  (set-face-attribute 'tab-bar-tab nil 
                      :background 'unspecified
                      :inherit 'default 
                      :weight 'bold
                      :box nil)

  ;; 3. Inactive tabs: inherit from 'shadow' (which all themes map to a faded grey)
  (set-face-attribute 'tab-bar-tab-inactive nil 
                      :background 'unspecified
                      :inherit 'shadow 
                      :weight 'normal
                      :box nil)
#+end_src

* Tabs
#+begin_src emacs-lisp
  ;; 1. Configure the native Tab Bar
  (use-package tab-bar
    :ensure nil
    :custom
    (tab-bar-show 1)                        ; Only show the bar if > 1 tab is open
    (tab-bar-close-button-show nil)         ; Hide the [X] for a cleaner UI
    (tab-bar-new-tab-choice 'list-bookmarks)    ; New tabs start clean
    (tab-bar-tab-hints t)
    (tab-bar-format '(tab-bar-format-tabs 
                      tab-bar-format-align-right 
                      tab-bar-format-global))
    :config
    (tab-bar-mode 1))

  ;; 2. Tabspaces: Isolate buffers per tab (The "Workspace" magic)
  (use-package tabspaces
    :ensure t
    ;; Filter consult-buffer to only show buffers in the current tab's workspace
    :hook (after-init . tabspaces-mode)
    :custom
    (tabspaces-use-filtered-buffers-as-default t)
    (tabspaces-default-tab "Default")
    (tabspaces-remove-to-default t)
    (tabspaces-include-buffers '("*scratch*"))
    ;; Tell project.el to open new projects in a brand new tab automatically
    (project-switch-commands '((project-find-file "Find file")
                               (project-dired "Dired")
                               (project-vc-dir "VC-Dir")
                               (project-eshell "Eshell")))
    :config
    (defalias 'project-other-tab-command 'tabspaces-open-or-create-project-and-workspace))
#+end_src
