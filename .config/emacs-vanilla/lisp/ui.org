* Fonts & Faces
Base faces only. The org-specific fixed-pitch overrides (tables, code
blocks, etc) live in =org-config.org=, next to the rest of the org
appearance settings.

The default font is set via =default-frame-alist= so new GUI frames get
their exact geometry /before/ mapping (preventing daemon resize flicker
and =olivetti-mode= margin miscalculations). The remaining face overrides
are hooked to =enable-theme-functions= and frame creation, scoped to
graphical frames, and leave =:box= untouched so =spacious-padding= can
manage tab-bar padding cleanly.
#+begin_src emacs-lisp
(add-to-list 'default-frame-alist '(font . "Iosevka-12:weight=medium"))

(defun my/apply-face-overrides (&optional frame)
  "(Re)apply font and tab-bar face overrides after theme load or frame creation."
  (let ((frame (or (and (framep frame) frame) (selected-frame))))
    (when (display-graphic-p frame)
      (with-selected-frame frame
        (set-face-attribute 'fixed-pitch frame
                            :family "Iosevka"
                            :height 1.0
                            :weight 'medium)

        (set-face-attribute 'variable-pitch frame
                            :family "Montserrat"
                            :height 1.08
                            :weight 'semi-bold)

        ;; Leave :box untouched; spacious-padding manages :box on tab-bar.
        (set-face-attribute 'tab-bar frame
                            :background 'unspecified
                            :inherit 'mode-line-inactive)

        ;; During daemon startup, `link' may not have a resolved foreground.
        ;; The underline's :color must be a color string or `foreground-color',
        ;; never `unspecified'.
        (let ((link-color (face-attribute 'link :foreground frame t)))
          (set-face-attribute 'tab-bar-tab frame
                              :inherit 'mode-line
                              :weight 'bold
                              :underline (list :color (if (stringp link-color)
                                                          link-color
                                                        'foreground-color)
                                               :style 'line
                                               :position t)))

        (set-face-attribute 'tab-bar-tab-inactive frame
                            :background 'unspecified
                            :inherit 'shadow
                            :weight 'normal)))))

;; Register before the initial theme load.
(add-hook 'enable-theme-functions #'my/apply-face-overrides)
(if (daemonp)
    (add-hook 'server-after-make-frame-hook #'my/apply-face-overrides)
  (add-hook 'window-setup-hook #'my/apply-face-overrides))

;; Load theme BEFORE spacious-padding so spacious-padding wraps the final faces.
(load-theme 'spacegrey-light t)
#+end_src

* Padding
Loaded after the theme and =nerd-icons= so it wraps the final face
definitions without fighting =my/apply-face-overrides=.
#+begin_src emacs-lisp
(use-package spacious-padding
  :ensure t
  :after nerd-icons
  :custom
  (spacious-padding-widths
   '( :internal-border-width 15
      :header-line-width 4
      :mode-line-width 4
      :tab-width 4
      :right-divider-width 30
      :scroll-bar-width 8))
  (spacious-padding-subtle-frame-lines (not (daemonp)))
  :config
  (spacious-padding-mode 1))
#+end_src

* Tabs & Workspaces
=tab-bar= and =project= are configured first so =tabspaces= can take
ownership of =tab-bar-mode= activation cleanly on startup.
#+begin_src emacs-lisp
;; 1. Configure the native Tab Bar
(use-package tab-bar
  :ensure nil
  :custom
  (tab-bar-show 1)                        ; Only show the bar if > 1 tab is open
  (tab-bar-close-button-show nil)         ; Hide the [X] for a cleaner UI
  (tab-bar-new-tab-choice "*scratch*")    ; New tabs start clean
  (tab-bar-tab-hints t)
  (tab-bar-format '(tab-bar-format-tabs
                    tab-bar-format-align-right
                    tab-bar-format-global)))

;; 2. Configure project.el commands before tabspaces aliases them
(use-package project
  :ensure nil
  :custom
  (project-switch-commands '((project-find-file "Find file")
                             (project-dired "Dired")
                             (project-vc-dir "VC-Dir")
                             (project-eshell "Eshell"))))

;; 3. Tabspaces: Isolate buffers per tab (The "Workspace" magic)
(use-package tabspaces
  :ensure t
  :after (tab-bar project)
  :hook (after-init . tabspaces-mode)
  :custom
  (tabspaces-session t)                   ; Save sessions automatically
  (tabspaces-session-auto-save-delay 300) ; Save after 5 idle minutes
  (tabspaces-session-auto-restore nil)
  (tabspaces-use-filtered-buffers-as-default t)
  (tabspaces-default-tab "Default")
  (tabspaces-remove-to-default t)
  (tabspaces-include-buffers '("*scratch*"))
  :config
  (tab-bar-mode 1)
  (defalias 'project-other-tab-command #'tabspaces-open-or-create-project-and-workspace))
#+end_src
