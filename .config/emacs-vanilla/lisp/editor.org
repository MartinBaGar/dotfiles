      ;; Add the Tempel Capf to `completion-at-point-functions'. `tempel-expand'

      ;; programming mode Capf, so it's tried first.
      (setq-local completion-at-point-functions
    		(cons #'tempel-expand completion-at-point-functions))
      (setopt tempel-path (concat (expand-file-name "templates" user-emacs-directory) "/*"))
      )

    (add-hook 'prog-mode-hook 'tempel-setup-capf)
    (add-hook 'text-mode-hook 'tempel-setup-capf))

  (use-package tempel-collection)

  (defun tempel-case-match (elt fields)
    (pcase elt
      (`(c ,text)
       (let ((at-start (save-excursion
                         (skip-chars-backward " \t")
                         (or (bobp)
                             (bolp)
                             (memq (char-before) '(?. ?? ?!))
                             (looking-back "^\\s-*[*\\-]+" (line-beginning-position))))))
         (if (and at-start (> (length text) 0))
             (concat (upcase (substring text 0 1)) (substring text 1))
           text)))))

  (defun my/open-tempel-templates ()
    (interactive)
    (find-file (f-dirname tempel-path)))

    :ensure t
    :config
    (key-chord-mode 1)

    ;; trigger accidentally when you are typing fast. (0.15 is a sweet spot).
    (setq key-chord-two-keys-delay 0.15)
    
    (with-eval-after-load 'meow
      ;; Only bind 'fd' to exit insert mode, and ONLY when in insert mode.
      (key-chord-define meow-insert-state-keymap "fd" #'meow-insert-exit)))

  (defun my/meow-surround ()
    "Surround the current active region with a pair of characters."
    (interactive)
    (if (region-active-p)
        (let* ((char (read-char "Surround with: "))
               ;; Map opening/closing brackets to their pairs. 
               ;; Any other character (like quotes or asterisks) is paired with itself.
               (pair (pcase char
                       ((or ?\( ?\)) '(?\( . ?\)))
                       ((or ?\[ ?\]) '(?\[ . ?\]))
                       ((or ?\{ ?\}) '(?\{ . ?\}))
                       ((or ?\< ?\>) '(?\< . ?\>))
                       (_ (cons char char))))
               (beg (region-beginning))
               (end (region-end)))
          (save-excursion
            (goto-char end)
            (insert (cdr pair))
            (goto-char beg)
            (insert (car pair)))
          ;; Drop back to normal state after wrapping
          (meow-cancel-selection))
      (message "No active region to surround.")))

  (defun my/copy-file-path (&optional arg)
    "Copy the current buffer's file path to the clipboard.
  If in a Dired/Dirvish buffer, copy the path of the file at point.
  If the buffer is not saved to a file, copy the default directory.
  With a prefix argument (e.g., `SPC 1`), copy ONLY the filename instead of the full path."
    (interactive "P")
    (let ((path (if (derived-mode-p 'dired-mode)
                    (dired-get-filename nil t)
                  (or buffer-file-name default-directory))))
      (if (not path)
          (message "No file path associated with this buffer.")
        (let ((text (if arg (file-name-nondirectory path) path)))
          (kill-new text)
          (message "Copied path: %s" text)))))

  (with-eval-after-load 'meow
    (defvar-keymap my/file-map
      :doc "Files"
      "f" #'find-file             ; Standard file search
      "s" #'save-buffer           ; Save current file
      "y" #'my/copy-file-path)    ; Our custom copy-path function

    (meow-leader-define-key 
     (cons "f" my/file-map)))
