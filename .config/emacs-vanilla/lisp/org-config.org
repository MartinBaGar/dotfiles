* Appearance (org-modern)
org-modern owns stars, tags, lists and checkboxes on its own -- no
separate bullets package needed alongside it.
#+begin_src emacs-lisp
(use-package org-modern
  :custom
  (org-modern-star '("◉" "○" "●" "○" "●"))
  ;; Disable org-modern table styling if you prefer standard Org tables
  (org-modern-table nil)
  (org-modern-list
   '(;; (?- . "-")
     (?* . "•")
     (?+ . "‣")))
  :hook
  (org-mode . org-modern-mode)
  (org-agenda-finalize . org-modern-agenda))

;; Source blocks and indentation structure
(use-package org-modern-indent
  :ensure (org-modern-indent :host github :repo "jdtsmith/org-modern-indent")
  :config
  ;; The '90' ensures this hook runs very late, after org-modern and org-indent
  (add-hook 'org-mode-hook #'org-modern-indent-mode 90))
#+end_src

* Fonts: prose vs. code
These faces are set globally, once -- not inside a per-buffer hook,
since `set-face-attribute' without a frame argument changes the face
everywhere, and re-running it on every org buffer you open is pure
waste. Only the buffer-local minor mode (`variable-pitch-mode') needs
to live in the hook.
#+begin_src emacs-lisp
(with-eval-after-load 'org
  (dolist (face '(org-table org-code org-verbatim org-block
                   org-meta-line org-drawer org-special-keyword
                   org-property-value org-document-info-keyword
                   org-date org-tag))
    (set-face-attribute face nil :inherit 'fixed-pitch)))

(add-hook 'org-mode-hook #'variable-pitch-mode)
#+end_src

* Org-roam
#+begin_src emacs-lisp
(use-package org-roam
  :ensure t
  :custom
  ;; Replace this string with the exact path to your Syncthing directory
  (org-roam-directory (file-truename "~/org/roam"))
  
  ;; Enables completion for node insertion anywhere
  (org-roam-completion-everywhere t)
  :config
  ;; This is the magic line. It automatically parses all your existing files
  ;; and builds the SQLite database so everything links up immediately.
  (org-roam-db-autosync-mode 1))

(with-eval-after-load 'meow
  ;; Recreate Doom's Notes submap -- SPC n ...
  (defvar-keymap my/notes-map
    :doc "Org-Roam / Notes"
    "f" #'org-roam-node-find       ; Find or create a node
    "i" #'org-roam-node-insert     ; Insert a link to a node
    "c" #'org-roam-capture         ; Capture a new node instantly
    "l" #'org-roam-buffer-toggle   ; Toggle the backlinks buffer
    "t" #'org-roam-tag-add)        ; Add a tag to the current node
    
  (meow-leader-define-key (cons "n" my/notes-map)))
#+end_src
