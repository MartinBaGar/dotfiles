  (defun my/prettify-string (str)
    "Convert STR into a character or composition sequence for `prettify-symbols-alist'."
    (let ((chars (string-to-list str)))
      (if (cdr chars)
          (cons (car chars)
                (mapcan (lambda (c) (list '(Br . Bl) c)) (cdr chars)))
        (car chars))))

  (defun my/org-lang-icon (lang)
    "Return the Nerd Font glyph string for Org source language LANG."
    (let* ((default-icon (nerd-icons-icon-for-mode 'fundamental-mode))
           (mode (and lang (org-src-get-lang-mode lang)))
           (base-mode (and mode
                           (intern (replace-regexp-in-string
                                    "-ts-mode\\'" "-mode" (symbol-name mode)))))
           (icon (and mode (nerd-icons-icon-for-mode mode)))
           (icon (if (and icon (equal icon default-icon))
                     (nerd-icons-icon-for-mode base-mode)
                   icon)))
      (if (and (stringp icon)
               (not (equal icon default-icon)))
          icon
        (nerd-icons-codicon "nf-cod-code"))))

  (defun my/org-src-prettify-alist ()
    "Build `prettify-symbols-alist' with Nerd Font icons for Org blocks."
    (let* ((langs '("python" "bash" "sh" "emacs-lisp" "elisp" "json" "yaml"
                    "toml" "typst" "c" "cpp" "rust" "js" "typescript" "svelte"
                    "html" "css" "sql" "awk" "makefile" "dockerfile" "xml"
                    "pyops"))
           (end-icon (my/prettify-string (nerd-icons-codicon "nf-cod-code")))
           (alist `(("#+end_src"     . ,end-icon)
                    ("#+END_SRC"     . ,end-icon)
                    ("#+begin_quote" . ,(my/prettify-string "“"))
                    ("#+BEGIN_QUOTE" . ,(my/prettify-string "“"))
                    ("#+end_quote"   . ,(my/prettify-string "”"))
                    ("#+END_QUOTE"   . ,(my/prettify-string "”"))
                    ("#+RESULTS:"    . ,(my/prettify-string "")))))
      (dolist (lang langs alist)
        (let ((composed (my/prettify-string (concat (my/org-lang-icon lang) " " lang))))
          (push (cons (concat "#+begin_src " lang) composed) alist)
          (push (cons (concat "#+BEGIN_SRC " lang) composed) alist)))))

  (defun my/org-setup-prettify-symbols ()
    "Enable prettified block headers and icons in Org buffers."
    (setq-local prettify-symbols-unprettify-at-point 'right-edge)

    (when (bound-and-true-p prettify-symbols-mode)
      (prettify-symbols-mode -1))
    (prettify-symbols-mode 1))

    (with-current-buffer buf

        (my/org-setup-prettify-symbols)))))

(defun my-dynamic-export-exclude-tags (info backend)
  "Dynamically tell Org which tags to ignore based on the backend."
  (cond
   ((eq backend 'typst)
    (plist-put info :exclude-tags (cons "html_only" org-export-exclude-tags)))
   ((eq backend 'html)
    (plist-put info :exclude-tags (cons "pdf_only" org-export-exclude-tags))))
  info)

;; IMPORTANT: Attach to filter-options-functions (which provides 2 arguments)
(add-hook 'org-export-filter-options-functions #'my-dynamic-export-exclude-tags)

;; 3. Conditional TOC Function
(defun my-org-export-conditional-toc (info backend)
  "Dynamically enable/disable TOC based on backend and ADTOC property."
  (message "export conditional toc has been called")
  (let ((my-custom-adtoc-value (plist-get info :adtoc)))
    (if my-custom-adtoc-value
        (cond
         ((eq backend 'html)
          (plist-put info :with-toc nil))
         ((eq backend 'typst)
          (plist-put info :with-toc t)))))
  info)

;; Attach to filter-options-functions
(add-hook 'org-export-filter-options-functions #'my-org-export-conditional-toc)

;; 4. Ignore Headlines Function (AST)
(defun my-org-export-ignore-headlines (data backend info)
  "Remove headlines tagged 'ignore', retain contents, and promote children.
Operates directly on the Org AST."
  (org-element-map data 'headline
    (lambda (object)
      (when (member "ignore" (org-element-property :tags object))
        (let ((level-top (org-element-property :level object))
              level-diff)
          (mapc (lambda (el)
                  ;; Recursively promote all nested child headlines
                  (org-element-map el 'headline
                    (lambda (el)
                      (when (equal 'headline (org-element-type el))
                        (unless level-diff
                          (setq level-diff (- (org-element-property :level el)
                                              level-top)))
                        (org-element-put-property el
                                                  :level (- (org-element-property :level el)
                                                            level-diff)))))
                  ;; Insert the contents back into the parse tree
                  (org-element-insert-before el object))
                (org-element-contents object)))
        ;; Remove the original headline node
        (org-element-extract-element object)))
    info nil)
  data)

;; Add it to the AST parsing hook
(add-hook 'org-export-filter-parse-tree-functions #'my-org-export-ignore-headlines)

(use-package ox-typst
  :after org
  :config
  ;; 1. OVERWRITE the package's broken example-block function
  ;; (This safely replaces the original function in memory)
  (defun org-typst-example-block (example-block contents info)
    "Transcode an EXAMPLE-BLOCK element from Org to Typst."
    (let ((value (org-element-property :value example-block)))
      (format "```\n%s```\n" value)))

  ;; 2. Define the TODO preservation hook
  (defun org-typst-preserve-todos (tree backend info)
    "Force TODO keywords into the headline title so ox-typst doesn't drop them."
    (when (eq backend 'typst)
      (org-element-map tree 'headline
        (lambda (hl)
          (let ((todo (org-element-property :todo-keyword hl)))
            (when todo
              (let ((title (org-element-property :title hl)))
                (org-element-put-property 
                 hl :title 
                 (cons (format "%s " todo) title))))))))
    tree)

  ;; 3. Define the Source Block Backslash fix
  (defun org-typst-fix-src-backslashes (tree backend info)
    "Escape backslashes in source blocks so Typst doesn't evaluate them."
    (when (eq backend 'typst)
      (org-element-map tree 'src-block
        (lambda (src)
          (let ((value (org-element-property :value src)))
            (when value
              (org-element-put-property 
               src :value 
               (replace-regexp-in-string "\\\\" "\\\\\\\\" value)))))))
    tree)

  ;; 5. FIX the broken inlinetask function signature
  (setq org-typst-format-inlinetask-function
        (lambda (todo todo-type priority title tags contents info)
          ;; Formats the inline task as a styled block with the title and contents
          (format "#block(fill: rgb(\"f4f4f4\"), stroke: 1pt + rgb(\"d3d3d3\"), inset: 8pt, radius: 4pt)[*%s %s*\n\n%s]"
                  (if todo todo "")
                  title
                  (or contents ""))))

  ;; 4. Attach the AST fixes to the Org export pipeline
  (add-to-list 'org-export-filter-parse-tree-functions #'org-typst-preserve-todos)
  (add-to-list 'org-export-filter-parse-tree-functions #'org-typst-fix-src-backslashes))
