* Appearance (org-modern)
org-modern owns stars, tags, lists and checkboxes on its own -- no
separate bullets package needed alongside it.
#+begin_src emacs-lisp
(use-package org-modern
  :custom
  (org-modern-star '("◉" "○" "●" "○" "●"))
  ;; Disable org-modern table styling if you prefer standard Org tables
  (org-modern-table nil)
  (org-modern-list
   '(;; (?- . "-")
     (?* . "•")
     (?+ . "‣")))
  :hook
  (org-mode . org-modern-mode)
  (org-agenda-finalize . org-modern-agenda))

;; Source blocks and indentation structure

  :ensure (org-modern-indent :host github :repo "jdtsmith/org-modern-indent")

  (add-hook 'org-mode-hook #'org-modern-indent-mode 90))
#+end_src

* Fonts: prose vs. code
These faces are set globally, once -- not inside a per-buffer hook,
since `set-face-attribute' without a frame argument changes the face
everywhere, and re-running it on every org buffer you open is pure
waste. Only the buffer-local minor mode (`variable-pitch-mode') needs
to live in the hook.
#+begin_src emacs-lisp
(with-eval-after-load 'org
  (dolist (face '(org-table org-code org-verbatim org-block
                   org-meta-line org-drawer org-special-keyword
                   org-property-value org-document-info-keyword
                   org-date org-tag))
    (set-face-attribute face nil :inherit 'fixed-pitch)))

(add-hook 'org-mode-hook #'variable-pitch-mode)
#+end_src

** Org-babel
#+begin_src emacs-lisp
  (with-eval-after-load 'org
    (org-babel-do-load-languages
     'org-babel-load-languages
     '((shell . t)
       (python . t)    ;; Added this since you use Python blocks
       (emacs-lisp . t))))

  (setq org-confirm-babel-evaluate nil)

  (with-eval-after-load 'ob-core
    (require 'cl-lib)

    (defun my/org-babel-fold-result (result &optional result-params info &rest _)
      "Fold the results drawer/block that was just written, sync or async,
  but only for blocks whose effective header args include `:fold yes'."
      (when (derived-mode-p 'org-mode)
        (let* ((params (and info (nth 2 info)))
               (fold   (cdr (assq :fold params))))
          (when (member fold '("yes" "t"))
            (save-excursion
              (let ((loc (ignore-errors (org-babel-where-is-src-block-result))))
                (when loc
                  (goto-char loc)
                  (let ((end (ignore-errors (org-babel-result-end))))
                    (when end
                      ;; Purge ANY existing hide-overlay touching this result
                      ;; block, alive or not. Comint async can re-run this
                      ;; advice several times as output streams in, and Org
                      ;; doesn't clean up the previous fold overlay before
                      ;; the next chunk lands, leaving a stale/shrunk
                      ;; overlay behind.
                      (setq org-babel-hide-result-overlays
                            (cl-remove-if
                             (lambda (ov)
                               (when (and (overlay-buffer ov)
                                          (>= (overlay-start ov) (1- loc))
                                          (<= (overlay-end ov) (1+ end)))
                                 (delete-overlay ov)
                                 t))
                             org-babel-hide-result-overlays)))
                    (org-babel-hide-result-toggle t)))))))))

    (advice-add 'org-babel-insert-result :after #'my/org-babel-fold-result))
#+end_src
