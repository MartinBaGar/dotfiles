* Fonts: prose vs. code
These faces are set globally, once -- not inside a per-buffer hook,
since `set-face-attribute' without a frame argument changes the face
everywhere, and re-running it on every org buffer you open is pure
waste. Only the buffer-local minor mode (`variable-pitch-mode') needs
to live in the hook.
#+begin_src emacs-lisp
(with-eval-after-load 'org
  (dolist (face '(org-table org-code org-verbatim org-block
                   org-meta-line org-drawer org-special-keyword
                   org-property-value org-document-info-keyword
                   org-date org-tag))
    (set-face-attribute face nil :inherit 'fixed-pitch)))

(add-hook 'org-mode-hook #'variable-pitch-mode)
#+end_src

** Org-babel
#+begin_src emacs-lisp
  (with-eval-after-load 'org
    (org-babel-do-load-languages
     'org-babel-load-languages
     '((shell . t)
       (python . t)    ;; Added this since you use Python blocks
       (emacs-lisp . t))))

  (setq org-confirm-babel-evaluate nil)

  (with-eval-after-load 'ob-core
    (require 'cl-lib)

    (defun my/org-babel-fold-result (result &optional result-params info &rest _)
      "Fold the results drawer/block that was just written, sync or async,
  but only for blocks whose effective header args include `:fold yes'."
      (when (derived-mode-p 'org-mode)
        (let* ((params (and info (nth 2 info)))
               (fold   (cdr (assq :fold params))))
          (when (member fold '("yes" "t"))
            (save-excursion
              (let ((loc (ignore-errors (org-babel-where-is-src-block-result))))
                (when loc
                  (goto-char loc)
                  (let ((end (ignore-errors (org-babel-result-end))))
                    (when end
                      ;; Purge ANY existing hide-overlay touching this result
                      ;; block, alive or not. Comint async can re-run this
                      ;; advice several times as output streams in, and Org
                      ;; doesn't clean up the previous fold overlay before
                      ;; the next chunk lands, leaving a stale/shrunk
                      ;; overlay behind.
                      (setq org-babel-hide-result-overlays
                            (cl-remove-if
                             (lambda (ov)
                               (when (and (overlay-buffer ov)
                                          (>= (overlay-start ov) (1- loc))
                                          (<= (overlay-end ov) (1+ end)))
                                 (delete-overlay ov)
                                 t))
                             org-babel-hide-result-overlays)))
                    (org-babel-hide-result-toggle t)))))))))

    (advice-add 'org-babel-insert-result :after #'my/org-babel-fold-result))
#+end_src

** Org-elabftw
#+begin_src emacs-lisp
(use-package elabftw-sync
  :ensure (:host github :repo "MartinBaGar/elabftw-sync")
  :custom
  (elabftw-sync-pdf-export-function 'org-typst-export-to-pdf))

;; 1. Register ADTOC
(with-eval-after-load 'ox
  (add-to-list 'org-export-options-alist
               '(:adtoc "ADTOC" "adtoc" nil t)))

;; 2. Exclude Tags Function
(defun my-dynamic-export-exclude-tags (info backend)
  "Dynamically tell Org which tags to ignore based on the backend."
  (cond
   ((eq backend 'typst)
    (plist-put info :exclude-tags (cons "html_only" org-export-exclude-tags)))
   ((eq backend 'html)
    (plist-put info :exclude-tags (cons "pdf_only" org-export-exclude-tags))))
  info)

;; IMPORTANT: Attach to filter-options-functions (which provides 2 arguments)
(add-hook 'org-export-filter-options-functions #'my-dynamic-export-exclude-tags)

;; 3. Conditional TOC Function
(defun my-org-export-conditional-toc (info backend)
  "Dynamically enable/disable TOC based on backend and ADTOC property."
  (message "export conditional toc has been called")
  (let ((my-custom-adtoc-value (plist-get info :adtoc)))
    (if my-custom-adtoc-value
        (cond
         ((eq backend 'html)
          (plist-put info :with-toc nil))
         ((eq backend 'typst)
          (plist-put info :with-toc t)))))
  info)

;; Attach to filter-options-functions
(add-hook 'org-export-filter-options-functions #'my-org-export-conditional-toc)

;; 4. Ignore Headlines Function (AST)
(defun my-org-export-ignore-headlines (data backend info)
  "Remove headlines tagged 'ignore', retain contents, and promote children.
Operates directly on the Org AST."
  (org-element-map data 'headline
    (lambda (object)
      (when (member "ignore" (org-element-property :tags object))
        (let ((level-top (org-element-property :level object))
              level-diff)
          (mapc (lambda (el)
                  ;; Recursively promote all nested child headlines
                  (org-element-map el 'headline
                    (lambda (el)
                      (when (equal 'headline (org-element-type el))
                        (unless level-diff
                          (setq level-diff (- (org-element-property :level el)
                                              level-top)))
                        (org-element-put-property el
                                                  :level (- (org-element-property :level el)
                                                            level-diff)))))
                  ;; Insert the contents back into the parse tree
                  (org-element-insert-before el object))
                (org-element-contents object)))
        ;; Remove the original headline node
        (org-element-extract-element object)))
    info nil)
  data)

;; Add it to the AST parsing hook
(add-hook 'org-export-filter-parse-tree-functions #'my-org-export-ignore-headlines)
